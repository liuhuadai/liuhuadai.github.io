\section{SUMMARY}
Researcher working on multimodal foundation models that perceive, reason, and generate: audio language models for speech, sound, and music; multimodal LLM reasoning; post-training with reinforcement learning, on-policy distillation, and chain-of-thought; and the latent representations and few-step samplers beneath diffusion and flow-based generation. 16 first-author papers at NeurIPS, ICML, ICLR, ACL, EMNLP, and ACM MM.

\section{RESEARCH HIGHLIGHTS}
\researchgroup{Audio Language Models}
\begin{itemize}
  \item \textbf{AudioCALM} (NeurIPS 2026). One continuous autoregressive model for speech, sound, and music: a flow-matching head replaces the softmax and block-causal attention supports streaming; beats CosyVoice 3.0 on WER (2.0\% vs.\ 2.2\%) and TangoFlux on FAD (1.95 vs.\ 2.70).
\end{itemize}
\researchgroup{Multimodal LLMs: Reasoning \& Evaluation}
\begin{itemize}
  \item \textbf{ThinkSound} (NeurIPS 2025). Multimodal-LLM chain-of-thought binds visual events to acoustic conditions before flow-matching synthesis, and drives object-level refinement and instruction-based editing; FD 34.6 vs.\ 43.3 for MMAudio on VGGSound, with 2.8\texttimes\ faster inference.
  \item \textbf{X-Stream} (ECCV 2026, co-first author). First benchmark for multi-stream streaming understanding (4,220 QA pairs over 932 videos, 11 subtasks), analysing MLLMs as signal multiplexers; state-of-the-art MLLMs score about 50\%.
\end{itemize}
\researchgroup{Post-training \& Alignment}
\begin{itemize}
  \item \textbf{PrismAudio} (ICLR 2026). Four reasoning modules with matching rewards; Fast-GRPO's hybrid ODE--SDE sampling exceeds Flow-GRPO's final reward in 200 steps (vs.\ 600+); best on all four perceptual dimensions.
  \item \textbf{MosaicAudio} (EMNLP 2026). Distills four flow-matching teachers into one controllable student with per-axis on-policy supervision and a perceptual manifold anchor.
\end{itemize}
\researchgroup{Latent Representations \& Fast Sampling}
\begin{itemize}
  \item \textbf{STAR-VAE} (ICML 2026). Structure--texture-aligned latent channels with a CNN--Mamba backbone beat the Stable Audio Open VAE on all reconstruction metrics at equal latent rate; text-to-audio FD 89.2\,$\to$\,72.5.
  \item \textbf{FlashAudio / AudioLCM} (ACL 2025 Oral / ACM MM 2024). \textbf{Single-step generation at 400\texttimes\ real time}, beating diffusion baselines that use hundreds of steps; AudioLCM synthesizes in 2 steps at 333\texttimes\ real time.
\end{itemize}
