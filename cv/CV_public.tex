%---------- SELECTED RESEARCH ----------
\section*{Selected Research}

\noindent\textbf{AudioCALM: Continuous Autoregressive LM for Audio Generation} \hfill \textit{NeurIPS 2026 (under review)}
\begin{itemize}[leftmargin=1.3em, itemsep=0.1ex, topsep=0.2ex]
    \item Proposed Continuous Autoregressive Language Modeling (CALM): a flow-matching head replaces the LM softmax to predict rectified-flow velocities over continuous audio latents, with block-causal AR-Flow attention enabling streaming, arbitrary-length speech, sound, and music generation in one LLM backbone.
    \item Introduced A-MoME (Asymmetric Mixture-of-Modality-Experts) to address asymmetric local--global text--audio mismatch in joint training, matching modality-specific SOTA while surpassing prior unified baselines across all three domains.
\end{itemize}

\vspace{0.4ex}
\noindent\textbf{STAR-VAE: Structured Topology-Aware Regularization for Audio} \hfill \textit{ICML 2026}
\begin{itemize}[leftmargin=1.3em, itemsep=0.1ex, topsep=0.2ex]
    \item Formalized the Rate--Distortion--Regularity Trilemma in audio VAEs; proposed channel-indexed KL regularization aligning latent capacity with audio's spectral hierarchy.
    \item Combined STAR with a hybrid CNN--Mamba architecture and LLM-based Flow Matching, achieving SOTA reconstruction and generation on AudioCaps and Song-Describer.
\end{itemize}

\vspace{0.4ex}
\noindent\textbf{PrismAudio: Decomposed CoT and Multi-dimensional Rewards for V2A} \hfill \textit{ICLR 2026}
\begin{itemize}[leftmargin=1.3em, itemsep=0.1ex, topsep=0.2ex]
    \item First V2A framework integrating reinforcement learning with decomposed CoT planning across semantic, temporal, aesthetic, and spatial dimensions; proposed Fast-GRPO with hybrid ODE--SDE sampling to reduce training overhead.
    \item Released the AudioCanvas benchmark and achieved SOTA across all four perceptual dimensions on VGGSound and AudioCanvas.
\end{itemize}

\vspace{0.4ex}
\noindent\textbf{ThinkSound: Chain-of-Thought Reasoning for Audio Generation} \hfill \textit{NeurIPS 2025}
\begin{itemize}[leftmargin=1.3em, itemsep=0.1ex, topsep=0.2ex]
    \item First framework to integrate chain-of-thought reasoning from multimodal LLMs into video-to-audio generation; built the large-scale AudioCoT dataset bridging visual, textual, and acoustic reasoning.
\end{itemize}

\vspace{0.4ex}
\noindent\textbf{FlashAudio: Rectified Flows for Ultra-Fast Text-to-Audio Generation} \hfill \textit{ACL 2025 Oral (SAC Highlight)}
\begin{itemize}[leftmargin=1.3em, itemsep=0.1ex, topsep=0.2ex]
    \item First work to apply Rectified Flows to TTA, enabling single-step inference at \textbf{400$\times$ real time} on one RTX 4090 while surpassing strong diffusion baselines requiring hundreds of steps.
\end{itemize}
